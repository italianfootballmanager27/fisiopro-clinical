        // Navigazione tra le 8 Sezioni
        function switchSection(sectionId) {
            currentSection = sectionId;
            document.querySelectorAll('.content-section').forEach(el => el.classList.add('hidden'));
            document.getElementById(`section-${sectionId}`).classList.remove('hidden');

            document.querySelectorAll('.section-tab').forEach(btn => {
                btn.classList.remove('border-brand-500', 'bg-brand-50', 'text-brand-700');
                btn.classList.add('border-slate-200', 'bg-white', 'text-slate-700');
            });
            const activeBtn = document.getElementById(`btn-sec-${sectionId}`);
            activeBtn.classList.remove('border-slate-200', 'bg-white', 'text-slate-700');
            activeBtn.classList.add('border-brand-500', 'bg-brand-50', 'text-brand-700');
            
            filterContent();
        }

        function filterByDistrict(district) {
            currentDistrict = district;
            document.querySelectorAll('.filter-btn').forEach(btn => {
                btn.classList.remove('bg-brand-600', 'text-white', 'font-semibold', 'shadow-sm');
                btn.classList.add('bg-slate-100', 'text-slate-600', 'font-medium');
            });
            event.target.classList.remove('bg-slate-100', 'text-slate-600', 'font-medium');
            event.target.classList.add('bg-brand-600', 'text-white', 'font-semibold', 'shadow-sm');

            filterContent();
        }

        // Filtro specifico per categorie dei Trattamenti (Ortopedici, Neurologici, Pediatrici)
        function filterTrattamentiCat(cat) {
            currentTrattamentiCat = cat;
            document.querySelectorAll('.tratt-cat-btn').forEach(btn => {
                btn.classList.remove('bg-brand-600', 'text-white', 'font-bold', 'shadow-sm');
                btn.classList.add('bg-slate-200', 'text-slate-700', 'font-semibold');
            });
            const activeBtn = document.getElementById(`btn-tratt-${cat}`);
            activeBtn.classList.remove('bg-slate-200', 'text-slate-700', 'font-semibold');
            activeBtn.classList.add('bg-brand-600', 'text-white', 'font-bold', 'shadow-sm');

            filterContent();
        }

        function filterContent() {
            const query = document.getElementById('global-search').value.toLowerCase();

            // 1. Ortopediche
            const filteredOrto = ortopedicheData.filter(item => {
                const matchDist = currentDistrict === 'all' || item.district === currentDistrict;
                const matchQuery = item.title.toLowerCase().includes(query) || item.def.toLowerCase().includes(query) || item.district.toLowerCase().includes(query);
                return matchDist && matchQuery;
            });
            renderOrtopediche(filteredOrto);
            document.getElementById('orto-count').innerText = `${filteredOrto.length} Patologie`;

            // 2. Neurologiche
            const filteredNeuro = neurologicheData.filter(item => {
                const matchDist = currentDistrict === 'all' || item.district === currentDistrict || item.district === 'Generale';
                const matchQuery = item.title.toLowerCase().includes(query) || item.def.toLowerCase().includes(query) || item.meds.toLowerCase().includes(query) || item.rehab.toLowerCase().includes(query);
                return matchDist && matchQuery;
            });
            renderNeurologiche(filteredNeuro);
            document.getElementById('neuro-count').innerText = `${filteredNeuro.length} Patologie`;

            // 3. Test Clinici
            const filteredTests = testCliniciData.filter(item => {
                const matchDist = currentDistrict === 'all' || item.district === currentDistrict;
                const matchQuery = item.title.toLowerCase().includes(query) || item.purpose.toLowerCase().includes(query) || item.pathology.toLowerCase().includes(query);
                return matchDist && matchQuery;
            });
            renderTests(filteredTests);
            document.getElementById('tests-count').innerText = `${filteredTests.length} Test A-Z`;

            // 4. Trattamenti (con filtro Categoria Ortopedici / Neurologici / Pediatrici)
            const filteredTrattamenti = trattamentiData.filter(item => {
                const matchCat = currentTrattamentiCat === 'all' || item.category === currentTrattamentiCat;
                const matchDist = currentDistrict === 'all' || item.district === currentDistrict || item.district === 'Generale';
                const matchQuery = item.title.toLowerCase().includes(query) || 
                                   item.def.toLowerCase().includes(query) || 
                                   item.category.toLowerCase().includes(query) || 
                                   item.indications.toLowerCase().includes(query) ||
                                   item.goals.toLowerCase().includes(query);
                return matchCat && matchDist && matchQuery;
            });
            renderTrattamenti(filteredTrattamenti);

            // 5. Strumentale
            const filteredStrum = strumentaleData.filter(item => {
                const matchDist = currentDistrict === 'all' || item.district === currentDistrict;
                const matchQuery = item.title.toLowerCase().includes(query) || item.def.toLowerCase().includes(query);
                return matchDist && matchQuery;
            });
            renderStrumentale(filteredStrum);

            // 6. Anatomia
            const filteredAnat = anatomiaData.filter(item => {
                const matchDist = currentDistrict === 'all' || item.district === currentDistrict;
                const matchQuery = item.title.toLowerCase().includes(query) || item.innervation.toLowerCase().includes(query) || item.daily.toLowerCase().includes(query);
                return matchDist && matchQuery;
            });
            renderAnatomia(filteredAnat);
            document.getElementById('anatomia-count').innerText = `${filteredAnat.length} Muscoli A-Z`;

            // 7. Esercizi
            const filteredEserc = eserciziData.filter(item => {
                const matchDist = currentDistrict === 'all' || item.district === currentDistrict;
                const matchQuery = item.title.toLowerCase().includes(query) || item.target.toLowerCase().includes(query);
                return matchDist && matchQuery;
            });
            renderEsercizi(filteredEserc);
            document.getElementById('esercizi-count').innerText = `${filteredEserc.length} Esercizi A-Z`;

            // 8. Protocolli
            const filteredProt = protocolliData.filter(item => {
                const matchDist = currentDistrict === 'all' || item.district === currentDistrict;
                const matchQuery = item.title.toLowerCase().includes(query) || item.surgery.toLowerCase().includes(query);
                return matchDist && matchQuery;
            });
            renderProtocolli(filteredProt);
            document.getElementById('protocolli-count').innerText = `${filteredProt.length} Protocolli A-Z`;
        }

        // Logica Timer Clinico
        let timerSeconds = 45;
        let defaultSeconds = 45;
        let timerInterval = null;

        function updateTimerDisplay() {
            const mins = Math.floor(timerSeconds / 60).toString().padStart(2, '0');
            const secs = (timerSeconds % 60).toString().padStart(2, '0');
            document.getElementById('timer-display').innerText = `${mins}:${secs}`;
        }

        function setTimer(secs) {
            pauseTimer();
            defaultSeconds = secs;
            timerSeconds = secs;
            updateTimerDisplay();
            document.getElementById('timer-status').innerText = `Impostato a ${secs} secondi`;
        }

        function startTimer() {
            if (timerInterval) return;
            document.getElementById('timer-status').innerText = "Esecuzione in corso...";
            timerInterval = setInterval(() => {
                if (timerSeconds > 0) {
                    timerSeconds--;
                    updateTimerDisplay();
                } else {
                    pauseTimer();
                    document.getElementById('timer-status').innerText = "Tempo completato!";
                }
            }, 1000);
        }

        function pauseTimer() {
            clearInterval(timerInterval);
            timerInterval = null;
            document.getElementById('timer-status').innerText = "In pausa";
        }

        function resetTimer() {
            pauseTimer();
            timerSeconds = defaultSeconds;
            updateTimerDisplay();
            document.getElementById('timer-status').innerText = "Pronto";
        }
    </script>
